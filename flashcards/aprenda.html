Embaralhar os cartões

O algoritmo Fisher-Yates é o padrão para embaralhar arrays de forma uniforme.

script.js — shuffle
function embaralhar(arr) {
  // Cria cópia para não alterar o original
  const copia = [...arr];

  // Algoritmo Fisher-Yates
  for (let i = copia.length - 1; i > 0; i--) {
    // Índice aleatório entre 0 e i
    const j = Math.floor(Math.random() * (i + 1));

    // Troca as posições i e j
    [copia[i], copia[j]] = [copia[j], copia[i]];
  }

  return copia;
}

// Uso:
const embaralhados = embaralhar(cartoes);

Sessão de estudo no terminal

Loop pelos cartões, mostra a pergunta, aguarda Enter, mostra a resposta e registra acerto/erro.

memorycards.py — sessão
def estudar(baralho: Baralho) -> None:
    lista = baralho.cartoes[:]  # cópia
    random.shuffle(lista)

    print(f"📚 {baralho.nome} ({len(lista)} cartões)\n")

    acertos, erros = 0, 0

    for i, c in enumerate(lista, 1):
        print(f"🃏 Cartão {i}/{len(lista)}")
        print(f"Pergunta: {c.frente}")

        # Pausa até o usuário pressionar Enter
        input("Pressione Enter para ver a resposta...")
        print(f"✅ Resposta: {c.verso}")

        resp = input("Acertou? (s/n): ").strip().lower()

        if resp == 's':
            acertos += 1; c.acertos += 1
        else:
            erros   += 1; c.erros   += 1
        print()

    pct = round(acertos / len(lista) * 100)
    print(f"🎉 Resultado: {acertos} acertos, {erros} erros ({pct}%)")

Salvar e carregar em JSON

Persista os baralhos num arquivo .json para não perder os dados.

memorycards.py — JSON
def salvar(baralho: Baralho, arquivo: str = "baralho.json"):
    with open(arquivo, "w", encoding="utf-8") as f:
        # dataclasses.asdict converte para dicionário
        import dataclasses
        json.dump(dataclasses.asdict(baralho), f, ensure_ascii=False, indent=2)

def carregar(arquivo: str = "baralho.json") -> Baralho:
    with open(arquivo, encoding="utf-8") as f:
        data = json.load(f)

    # Reconstrói os objetos a partir do dicionário
    cartoes = [Cartao(**c) for c in data['cartoes']]
    return Baralho(nome=data['nome'], cartoes=cartoes)

          <p>{frente}</p>

        </div>

      </div>

2
Componente Sessao — navegação entre cartões

Usa dois estados: o índice atual e se o cartão está virado. Ao avançar, reseta a virada.

Sessao.jsx
import { useState } from 'react';
import { Cartao } from './Cartao';

export function Sessao({ cartoes }) {
  const [idx,    setIdx]    = useState(0);
  const [acertos, setAcertos] = useState(0);

  const atual = cartoes[idx];
  const fim   = idx >= cartoes.length;

  function responder(acertou) {
    if (acertou) setAcertos(a => a + 1);
    setIdx(i => i + 1); // avança para o próximo
  }

  if (fim) return (
    <div className="resultado">
      <h2>🎉 Fim da sessão!</h2>
      <p>{acertos} / {cartoes.length} acertos</p>
    </div>
  );

  return (
    <div>
      <p>{idx + 1} / {cartoes.length}</p>
      <Cartao frente={atual.frente} verso={atual.verso} />
      <button onClick={() => responder(false)}>❌ Errei</button>
      <button onClick={() => responder(true)}>✅ Acertei</button>
    </div>
  );
}
💡 Em React, nunca modifique o estado diretamente (virado = true ✗). Sempre use o setter (setVirado(true) ✓) para que o React saiba re-renderizar.

Criando as tabelas (DDL)

CREATE TABLE define a estrutura. Use FOREIGN KEY para vincular cartões ao seu baralho.

schema.sql
-- Tabela de baralhos
CREATE TABLE baralhos (
  id         INTEGER  PRIMARY KEY AUTOINCREMENT,
  nome       TEXT     NOT NULL,
  descricao  TEXT,
  criado_em  DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Tabela de cartões (1 baralho → N cartões)
CREATE TABLE cartoes (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  baralho_id  INTEGER NOT NULL,
  frente      TEXT    NOT NULL,
  verso       TEXT    NOT NULL,
  acertos     INTEGER DEFAULT 0,
  erros       INTEGER DEFAULT 0,

  -- Chave estrangeira: se o baralho for deletado, cartões também são
  FOREIGN KEY (baralho_id) REFERENCES baralhos(id)
    ON DELETE CASCADE
);
2
CRUD — Inserir, Ler, Atualizar, Deletar

As quatro operações fundamentais de qualquer banco de dados.

queries.sql
-- CREATE: inserir novo cartão
INSERT INTO cartoes (baralho_id, frente, verso)
VALUES (1, 'Capital do Brasil', 'Brasília');

-- READ: buscar todos os cartões de um baralho
SELECT * FROM cartoes
WHERE baralho_id = 1
ORDER BY RANDOM();    -- embaralha no próprio banco!

-- UPDATE: registrar acerto
UPDATE cartoes
SET    acertos = acertos + 1
WHERE  id = 42;

-- DELETE: remover um cartão
DELETE FROM cartoes WHERE id = 42;

JOIN e estatísticas de desempenho

JOIN combina tabelas. Aqui calculamos o aproveitamento de cada baralho.

relatorio.sql
-- Aproveitamento por baralho (acertos / total tentativas)
SELECT
  b.nome                              AS baralho,
  COUNT(c.id)                          AS total_cartoes,
  SUM(c.acertos)                      AS acertos,
  ROUND(
    SUM(c.acertos) * 100.0 /
    NULLIF(SUM(c.acertos + c.erros), 0),
    1
  )                                   AS aproveitamento_pct

FROM       baralhos b
INNER JOIN cartoes  c ON c.baralho_id = b.id

GROUP BY b.id
ORDER BY aproveitamento_pct DESC;
💡 Para apps pequenos use SQLite (arquivo local, sem servidor). Para produção com múltiplos usuários, use PostgreSQL ou MySQL.

Structs — o modelo de dados em Go

Em Go não há classes — usamos struct. A tag json:"..." controla como o campo aparece no JSON da API.

models.go
package main

// Struct define a estrutura do cartão
type Cartao struct {
  ID      int    `json:"id"`
  Frente  string `json:"frente"`
  Verso   string `json:"verso"`
  Acertos int    `json:"acertos"`
  Erros   int    `json:"erros"`
}

type Baralho struct {
  ID      int      `json:"id"`
  Nome    string   `json:"nome"`
  Cartoes []Cartao `json:"cartoes"`
}

// Método da struct: retorna aproveitamento
func (b Baralho) Aproveitamento() float64 {
  total, acertos := 0, 0
  for _, c := range b.Cartoes {
    total   += c.Acertos + c.Erros
    acertos += c.Acertos
  }
  if total == 0 { return 0 }
  return float64(acertos) / float64(total) * 100
}
2
API REST com net/http (sem framework)

Go tem HTTP nativo na stdlib. http.HandleFunc registra rotas, e json.NewEncoder serializa a resposta.

main.go — servidor
package main

import (
  "encoding/json"
  "net/http"
)

// Slice de baralhos (em produção viria do banco)
var baralhos = []Baralho{}

func listarBaralhos(w http.ResponseWriter, r *http.Request) {
  // Define o Content-Type da resposta
  w.Header().Set("Content-Type", "application/json")

  // Codifica o slice de baralhos como JSON e escreve na resposta
  json.NewEncoder(w).Encode(baralhos)
}

func main() {
  // Registra a rota GET /baralhos
  http.HandleFunc("/baralhos", listarBaralhos)

  // Inicia o servidor na porta 8080
  http.ListenAndServe(":8080", nil)
}
3
Goroutines — concorrência simples

Com go func() você executa tarefas em paralelo (ex: notificações de revisão) sem complexidade.

notificador.go
import ( "fmt"; "time" )

// Envia lembrete de revisão a cada 24h
func iniciarLembrete(usuario string) {
  // "go" inicia uma goroutine (thread leve)
  go func() {
    for {
      time.Sleep(24 * time.Hour)
      fmt.Printf("⏰ %s, hora de revisar seus cartões!\n", usuario)
    }
  }()
}

// O servidor principal não é bloqueado pela goroutine acima
func main() {
  iniciarLembrete("Maria")
  iniciarLembrete("João")
  // ... servidor continua rodando normalmente
  http.ListenAndServe(":8080", nil)
}

ViewModel + StateFlow (Android Jetpack)

ViewModel guarda o estado da UI sobrevivendo à rotação do celular. StateFlow emite atualizações reativamente.

CartaoViewModel.kt
import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.*

class CartaoViewModel: ViewModel() {

  // MutableStateFlow: estado interno (privado)
  private val _virado = MutableStateFlow(false)

  // StateFlow: versão pública, somente leitura
  val virado: StateFlow<Boolean> = _virado

  fun virar() {
    _virado.value = !_virado.value
  }

  fun responder(acertou: Boolean) {
    // Lógica de pontuação aqui
    _virado.value = false   // reseta para o próximo cartão
  }
}

// @Composable = função de UI (como componente React)
@Composable
fun CartaoCard(cartao: Cartao) {
  // remember + mutableStateOf = estado local do Composable
  var virado by remember { mutableStateOf(false) }

  Card(
    modifier = Modifier
      .fillMaxWidth()
      .clickable { virado = !virado }
  ) {
    Text(
      text = if (virado) cartao.verso else cartao.frente,
      style = MaterialTheme.typography.headlineMedium
    )
  }
}

Struct com protocol Identifiable

Swift usa struct para modelos de dados. Identifiable permite usar o cartão em listas SwiftUI.

Cartao.swift
import Foundation

// Identifiable: tem um "id" único (necessário para ForEach SwiftUI)
// Codable: pode ser serializado/deserializado de JSON
struct Cartao: Identifiable, Codable {
    let  id:      UUID = UUID()   // UUID gerado automaticamente
    var  frente:  String
    var  verso:   String
    var  acertos: Int = 0
    var  erros:   Int = 0

    // Propriedade computada: aproveitamento em %
    var aproveitamento: Double {
        let total = acertos + erros
        return total > 0 ? Double(acertos) / Double(total) * 100 : 0
    }
}
2
ObservableObject — estado compartilhado

@Published notifica a SwiftUI automaticamente quando o valor muda — equivalente ao useState do React.

BaralhoStore.swift
import Combine

// ObservableObject: a SwiftUI "observa" as mudanças
class BaralhoStore: ObservableObject {

  // @Published: qualquer mudança re-renderiza a View
  @Published var cartoes: [Cartao] = []
  @Published var indiceAtual: Int = 0

  var cartaoAtual: Cartao? {
    guard indiceAtual < cartoes.count else { return nil }
    return cartoes[indiceAtual]
  }

  func acertar() {
    cartoes[indiceAtual].acertos += 1
    indiceAtual += 1
  }

  func errar() {
    cartoes[indiceAtual].erros += 1
    indiceAtual += 1
  }
}
3
SwiftUI — View do cartão com animação 3D

.rotation3DEffect faz o efeito de virada nativo no iOS, e withAnimation cria a transição suave.

CartaoView.swift
import SwiftUI

struct CartaoView: View {
    let  cartao: Cartao

    // @State = estado local da View (como useState)
    @State private var virado = false
    @State private var angulo: Double = 0

    var body: some View {
        ZStack {
            // Mostra frente ou verso dependendo do ângulo
            if angulo < 90 {
                Text(cartao.frente).font(.title)
            } else {
                Text(cartao.verso).font(.title)
            }
        }
        // Efeito 3D nativo do SwiftUI
        .rotation3DEffect(.degrees(angulo), axis: (x: 0, y: 1, z: 0))
        .onTapGesture {
            // withAnimation aplica a transição suave
            withAnimation(.easeInOut(duration: 0.5)) {
                angulo = virado ? 0 : 180
                virado.toggle()
            }
        }
    }
}
