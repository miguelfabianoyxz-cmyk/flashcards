  // useEffect roda depois de montar o componente

    fetch('/api/posts')

  return (
    <ul>
      {posts.map(p => <li key={p.id}>{p.title}</li>)}
    </ul>
  );
}

Middleware de autenticação JWT

auth.py
from fastapi import Depends, Header
import jwt

SECRET = "chave-secreta"

def verificar_token(
    authorization: str = Header()
):
    token = authorization.removeprefix("Bearer ")
    try:
        payload = jwt.decode(
            token, SECRET,
            algorithms=["HS256"]
        )
        return payload["user_id"]
    except jwt.InvalidTokenError:
        raise HTTPException(401)

# Uso: rota protegida
@app.get("/perfil")
def perfil(uid=Depends(verificar_token)):
    return {"user_id": uid}

Go — Middleware de CORS + roteamento

server.go
package main
import (
  "encoding/json"; "net/http"
)

// Middleware: adiciona headers CORS
func cors(next http.Handler) http.Handler {
  return http.HandlerFunc(
    func(w http.ResponseWriter, r *http.Request) {
      w.Header().Set("Access-Control-Allow-Origin","*")
      next.ServeHTTP(w, r)
    }
  )
}
func main() {
  mux := http.NewServeMux()
  mux.HandleFunc("/api/ok",
    func(w http.ResponseWriter, r *http.Request) {
      json.NewEncoder(w).Encode(map[string]bool{"ok":true})
    }
  )
  http.ListenAndServe(":8080", cors(mux))
}

C — Alocação dinâmica de cartões

cartao.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
  char* frente;
  char* verso;
} Cartao;

/* Cria cartão na heap */
Cartao* novo_cartao(const char* f, const char* v) {
  Cartao* c = malloc(sizeof(Cartao));
  c->frente = strdup(f);
  c->verso  = strdup(v);
  return c;
}

/* SEMPRE libera a memória alocada */
void free_cartao(Cartao* c) {
  free(c->frente);
  free(c->verso);
  free(c);
}

Python — Script de importação de CSV

importar_csv.py
#!/usr/bin/env python3
import csv, json, sys
from pathlib import Path

def csv_para_baralho(arquivo_csv: str) -> dict:
    cartoes = []
    with open(arquivo_csv, encoding="utf-8") as f:
        # Detecta separador automaticamente
        leitor = csv.DictReader(f)
        for linha in leitor:
            cartoes.append({
                "frente": linha["frente"].strip(),
                "verso":  linha["verso"].strip(),
            })
    return {"nome": Path(arquivo_csv).stem,
            "cartoes": cartoes}

if __name__ == "__main__":
    baralho = csv_para_baralho(sys.argv[1])
    print(json.dumps(baralho, ensure_ascii=False, indent=2))

Observer — notificar progresso (TypeScript)

observer.ts
type Listener =(acertos: number) => void;

class SessaoEstudo {
  private acertos = 0;
  private listeners: Listener[] = [];

  // Inscreve observador
  subscribe(fn: Listener) {
    this.listeners.push(fn);
  }

  acertar() {
    this.acertos++;
    // Notifica todos os observadores
    this.listeners.forEach(fn => fn(this.acertos));
  }
}

// Uso: UI e analytics ouvem o mesmo evento
const sessao = new SessaoEstudo();
sessao.subscribe(n => ui.atualizar(n));
sessao.subscribe(n => analytics.log(n));

Factory — criação de cartões especializados

factory.py
from abc import ABC, abstractmethod

class Cartao(ABC):
    @abstractmethod
    def renderizar(self) -> str: ...

class CartaoTexto(Cartao):
    def renderizar(self): return "[texto]"

class CartaoImagem(Cartao):
    def renderizar(self): return "[imagem]"

# Factory: decide qual classe instanciar
def criar_cartao(tipo: str) -> Cartao:
    match tipo:
        case "texto":  return CartaoTexto()
        case "imagem": return CartaoImagem()
        case _:
            raise ValueError(f"Tipo '{tipo}' inválido")

│  PRESENTATION  │  CartaoController  (HTTP handlers, CLI)      │
│                │  Só recebe input → chama service → retorna   │
├──────────────────────────────────────────────────────────────┤
│  APPLICATION   │  CartaoService     (regras de negócio)       │
│                │  Orquestra operações, valida, transforma      │
├──────────────────────────────────────────────────────────────┤
│  DOMAIN        │  Cartao, Baralho   (entidades puras)         │
│                │  Sem dependência de frameworks ou I/O        │
├──────────────────────────────────────────────────────────────┤
│  INFRASTRUCTURE│  CartaoRepository  (banco, cache, APIs ext.) │
│                │  Implementa interfaces definidas no Domain    │
└──────────────────────────────────────────────────────────────┘
📐 SOLID em 5 segundos: S=uma responsabilidade por classe, O=aberto para extensão, L=subclasses substituem pai, I=interfaces específicas, D=dependa de abstrações, não implementações.
